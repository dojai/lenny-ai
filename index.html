import os import uuid import logging from typing import Literal from dotenv import load_dotenv from fastapi import FastAPI, HTTPException from fastapi.responses import HTMLResponse from openai import OpenAI from pydantic import BaseModel, Field # ========================================== # LENNY AI CONFIGURATION # ========================================== load_dotenv() APP_NAME = "LENNY AI" MODEL = os.getenv("LENNY_MODEL", "gpt-4.1-mini") API_KEY = os.getenv("OPENAI_API_KEY") logging.basicConfig(level=logging.INFO) logger = logging.getLogger("lenny-ai") client = OpenAI(api_key=API_KEY) if API_KEY else None app = FastAPI( title="LENNY AI API", description="The official LENNY AI platform", version="1.0.0" ) # ========================================== # LENNY AI WEBSITE # ========================================== HTML = r""" LENNY AI
L
LENNY AI
● AI PLATFORM

The future starts here.

Your intelligence partner. Ask anything.

✦
LENNY AI Your intelligent assistant
Hello! I'm LENNY AI. I'm here to help you explore ideas, answer questions, write code and learn. What would you like to discover today?
LENNY AI © 2026 · Intelligent conversations
""" # ========================================== # WEBSITE ROUTE # ========================================== @app.get("/", response_class=HTMLResponse) async def homepage(): return HTML # ========================================== # REQUEST VALIDATION # ========================================== class ChatMessage(BaseModel): role: Literal["user", "assistant"] content: str = Field(min_length=1, max_length=12000) class ChatRequest(BaseModel): messages: list[ChatMessage] = Field( min_length=1, max_length=20 ) # ========================================== # HEALTH CHECK # ========================================== @app.get("/health") async def health(): return { "name": APP_NAME, "status": "online", "ai_configured": client is not None } # ========================================== # MODEL INFORMATION # ========================================== @app.get("/v1/models") async def models(): return { "object": "list", "data": [ { "id": "lenny-smart", "object": "model", "owned_by": "lenny" } ] } # ========================================== # LENNY AI CHAT API # ========================================== @app.post("/v1/chat/completions") async def chat(request: ChatRequest): if client is None: raise HTTPException( status_code=503, detail="LENNY AI API key is not configured." ) clean_messages = [] for message in request.messages: content = message.content.strip() if not content: raise HTTPException( status_code=400, detail="Message cannot be empty." ) clean_messages.append({ "role": message.role, "content": content }) try: response = client.responses.create( model=MODEL, instructions=""" You are LENNY AI, an intelligent, helpful, friendly and professional AI assistant. Your identity is LENNY AI. Always introduce yourself as LENNY AI when asked about your identity. Answer questions accurately. Explain complicated topics clearly. Help with writing, learning, mathematics and coding. Maintain conversation context. Be honest about uncertainty. Never invent facts. """, input=clean_messages ) answer = response.output_text if not answer: raise ValueError("The AI returned an empty response.") return { "id": "lenny-" + str(uuid.uuid4()), "object": "chat.completion", "model": "lenny-smart", "choices": [ { "index": 0, "message": { "role": "assistant", "content": answer }, "finish_reason": "stop" } ] } except HTTPException: raise except Exception: logger.exception("LENNY AI API request failed") raise HTTPException( status_code=502, detail="LENNY AI encountered an engine error. Check server logs." ) # ========================================== # FAVICON # ========================================== @app.get("/favicon.ico") async def favicon(): return {"message": "LENNY AI"}
